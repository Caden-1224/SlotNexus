# SummerTTS 冒烟构建（剥离 tts_server 的 ZMQ+ALSA 集成层）。
# 与 tts/CMakeLists.txt（作者集成版，产物 tts_server）的区别：
#   - 目标 tts_smoke，入口 tts_smoke.cpp（非 tts_server/src/main.cpp）
#   - 不编译 tts_server/，不链接 ZMQ / ALSA / PortAudio
#   - 仅依赖 Eigen 3.4.0（本目录自带）+ -fopenmp
# 用法：把本文件作为板端工作目录的 CMakeLists.txt，mkdir build && cd build && cmake .. && make -j
# -include cstdint：上游 tts 的若干头（pinyinmap.h / hanzi2phoneid.h / Hanz2Piny.h）
#   使用 uint16_t 却未自给自足 include <cstdint>；GCC 13（板端 Ubuntu 24.04）不传递暴露，
#   故用 -include cstdint 全局补齐，不改动上游源码。
cmake_minimum_required(VERSION 3.5)
project(tts_smoke_proj CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -fopenmp -include cstdint")
